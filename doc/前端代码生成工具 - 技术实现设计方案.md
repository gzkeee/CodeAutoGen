# 前端代码生成工具 - 技术实现设计方案

## 1. 设计目标

基于已定稿的需求文档，实现一个**轻量级、可维护、可扩展**的代码生成工具。该工具将声明式的 JSON 配置编译为可直接运行的 React + Ant Design 代码。

### 1.1 核心设计原则

| 原则 | 说明 |
| :--- | :--- |
| **约定优于配置** | 通过组件类型自动推导数据流向行为，无需额外配置 `trigger` 或 `dataPass` |
| **模板即代码** | 使用模板字符串生成代码，配合 Prettier 格式化，无需引入繁重的 AST 框架 |
| **可读性优先** | 生成的代码是**可退出（Ejectable）**的，开发者可随时修改，不依赖工具 |
| **管道式架构** | 解析 → 建模 → 渲染 → 输出，各层职责清晰，易于测试和扩展 |

## 3. 核心数据模型

### 3.1 内部组件模型（ComponentModel）

解析完成后，每个组件会被转换为以下结构：

```javascript
// 组件内部模型
{
  id: 'SearchForm',           // 组件唯一标识
  type: 'Form',               // 'Form' | 'Table'
  fields: [...],              // 字段配置数组
  url: { load, query, save }, // 接口配置
  incoming: [],               // 谁给我传数据（组件ID数组）
  outgoing: [                 // 我把数据传给谁
    { target: 'ResultTable', behavior: 'formToTable' },
    { target: 'EditForm', behavior: 'tableToForm' }
  ],
  setValueAliases: {          // 自动生成的 setValue 别名映射
    'ResultTable': 'setValueResultTable',
    'EditForm': 'setValueEditForm'
  }
}
```

### 3.2 流向行为枚举（Behavior Enum）

```javascript
const BEHAVIORS = {
  FORM_TO_TABLE: 'formToTable',         // Form → Table
  TABLE_TO_FORM: 'tableToForm',         // Table → Form
  TABLE_TO_TABLE: 'tableToTable',       // Table → Table
  FORM_TO_FORM: 'formToForm',           // Form → Form
  CONTAINER_TO_CONTAINER: 'containerToContainer', // Container → Container
  CONTAINER_TO_FORM: 'containerToForm',           // Container → Form
  CONTAINER_TO_TABLE: 'containerToTable',         // Container → Table
  FORM_TO_CONTAINER: 'formToContainer',           // Form → Container
  TABLE_TO_CONTAINER: 'tableToContainer',         // Table → Container
};
```

### 3.3 流向推导规则表

| 源类型 | 目标类型 | Behavior | 触发事件 | 传递数据 | 目标反应 |
| :--- | :--- | :--- | :--- | :--- | :--- |
| Form | Table | `formToTable` | `onFinish` | `res.LIST` | 渲染为 `dataSource` |
| Table | Form | `tableToForm` | `onRowClick` | `record` | 执行 `form.setFieldsValue` |
| Table | Table | `tableToTable` | `onRowClick` | `record` | 提取 ID 触发 `load` |
| Form | Form | `formToForm` | `onSelectChange` | `selectedValue` | 触发 `load` 加载下拉选项 |
| Container | Container | `containerToContainer` | OUT→IN 透传 | 子组件 OUT 数据 | 通过 IN 注入目标子组件 |
| Container | Form | `containerToForm` | OUT→父级 setValue | 子组件 OUT 数据 | 执行 `form.setFieldsValue` |
| Container | Table | `containerToTable` | OUT→父级 setValue | 子组件 OUT 数据 | 渲染为 `dataSource` |
| Form | Container | `formToContainer` | `onFinish` | `res.LIST` | 通过 IN 注入子组件 |
| Table | Container | `tableToContainer` | `onRowClick` | `record` | 通过 IN 注入子组件 |

## 4. 核心模块实现方案

### 4.1 解析器 (Parser)

**功能**：读取 JSON 配置，进行基础校验，构建组件名映射表。

**实现要点**：

```javascript
// src/core/parser.js
function parseConfig(configPath) {
  const raw = fs.readJsonSync(configPath);
  const { components, DataFlow } = raw;
  
  // 1. 校验组件是否存在
  DataFlow.links.forEach(link => {
    if (!components[link.from]) throw new Error(`组件 ${link.from} 未定义`);
    if (!components[link.to]) throw new Error(`组件 ${link.to} 未定义`);
  });
  
  // 2. 构建组件映射表
  const componentMap = {};
  Object.keys(components).forEach(id => {
    componentMap[id] = {
      id,
      ...components[id],
      incoming: [],
      outgoing: []
    };
  });
  
  // 3. 构建拓扑关系（先构建出度，后续再构建入度）
  DataFlow.links.forEach(link => {
    componentMap[link.from].outgoing.push({
      target: link.to,
      behavior: getBehavior(componentMap[link.from].Type, componentMap[link.to].Type)
    });
  });
  
  return { componentMap, links: DataFlow.links };
}

// 流向推导函数
function getBehavior(fromType, toType) {
  if (fromType === 'Form' && toType === 'Table') return 'formToTable';
  if (fromType === 'Table' && toType === 'Form') return 'tableToForm';
  if (fromType === 'Table' && toType === 'Table') return 'tableToTable';
  if (fromType === 'Form' && toType === 'Form') return 'formToForm';
  throw new Error(`不支持的流向: ${fromType} → ${toType}`);
}
```

### 4.2 模型构建器 (ModelBuilder)

**功能**：完善组件模型，计算入度，生成 setValue 别名。

**实现要点**：

```javascript
// src/core/modelBuilder.js
function buildModel(componentMap, links) {
  // 1. 构建入度（计算谁给我传数据）
  links.forEach(link => {
    componentMap[link.to].incoming.push(link.from);
  });
  
  // 2. 为每个组件生成 setValue 别名
  Object.keys(componentMap).forEach(id => {
    const comp = componentMap[id];
    const outgoingCount = comp.outgoing.length;
    
    if (outgoingCount === 0) {
      comp.setValueAliases = {};
    } else if (outgoingCount === 1) {
      // 单个输出：使用默认名称 setValue
      comp.setValueAliases = { [comp.outgoing[0].target]: 'setValue' };
    } else {
      // 多个输出：生成别名 setValue[TargetName]
      comp.setValueAliases = {};
      comp.outgoing.forEach(out => {
        comp.setValueAliases[out.target] = `setValue${out.target}`;
      });
    }
  });
  
  return componentMap;
}
```

### 4.3 渲染器 (Renderer)

**功能**：遍历组件拓扑图，调用模板生成代码。

**实现要点**：渲染器采用**分层拼接**策略，先渲染原子字段，再组装组件，最后组装页面。

```javascript
// src/core/renderer.js
function render(componentMap, pageName) {
  const components = Object.keys(componentMap).map(id => {
    return renderComponent(id, componentMap[id]);
  });
  
  const indexContent = renderIndex(pageName, componentMap);
  const serviceContent = renderService(componentMap);
  
  return {
    index: indexContent,
    components: components,
    service: serviceContent
  };
}

// 渲染单个组件
function renderComponent(id, comp) {
  const fieldsCode = renderFields(comp.fields);
  const loadCode = renderLoad(comp.url.load);
  const queryCode = renderQuery(comp.url.query, comp.outgoing);
  const saveCode = renderSave(comp.url.save);
  const eventCode = renderEvents(comp);
  
  // 组合模板...
  return templateComponent({
    name: id,
    fields: fieldsCode,
    load: loadCode,
    query: queryCode,
    save: saveCode,
    events: eventCode
  });
}
```

### 4.4 事件绑定器 (EventBinder)

**功能**：根据流向类型，生成对应的事件绑定代码。

**实现要点**：

```javascript
// src/core/eventBinder.js
function bindEvents(comp) {
  const events = [];
  
  comp.outgoing.forEach(out => {
    const behavior = out.behavior;
    const alias = comp.setValueAliases[out.target];
    
    switch(behavior) {
      case 'formToTable':
        // 在 Form 的 onFinish 中调用
        events.push({
          prop: 'onFinish',
          code: `(values) => handleQuery(values, ${alias})`
        });
        break;
        
      case 'tableToForm':
      case 'tableToTable':
        // 在 Table 的 onRow 中调用
        events.push({
          prop: 'onRow',
          code: `(record) => ({ onClick: () => ${alias}(record) })`
        });
        break;
        
      case 'formToForm':
        // 在 Form 的 onValuesChange 中调用
        events.push({
          prop: 'onValuesChange',
          code: `(changed, all) => { if (changed.${comp.selectField}) ${alias}(changed.${comp.selectField}) }`
        });
        break;
    }
  });
  
  return events;
}
```

## 5. 模板系统设计

### 5.1 原子模板（字段级别）

```javascript
// src/templates/atoms.js
function renderField(field) {
  if (field.type === 'Input') {
    return `<Form.Item name="${field.name}" label="${field.label}">
      <Input placeholder="请输入${field.label}" ${field.disabled ? 'disabled' : ''} />
    </Form.Item>`;
  }
  
  if (field.type === 'Select') {
    const options = field.options.map(o => 
      `<Select.Option value="${o.value}">${o.label}</Select.Option>`
    ).join('');
    return `<Form.Item name="${field.name}" label="${field.label}">
      <Select placeholder="请选择${field.label}" ${field.disabled ? 'disabled' : ''}>
        ${options}
      </Select>
    </Form.Item>`;
  }
}
```

### 5.3 页面模板（index.jsx）

```javascript
// src/templates/index.js
function templateIndex({ pageName, components, componentMap }) {
  // 生成状态声明
  const states = Object.keys(componentMap).map(id => {
    const comp = componentMap[id];
    const initValue = comp.type === 'Table' ? '[]' : '{}';
    return `const [${id}Data, set${id}Data] = useState(${initValue});`;
  }).join('\n  ');
  
  // 生成组件渲染
  const renders = Object.keys(componentMap).map(id => {
    const comp = componentMap[id];
    const props = [];
    props.push(`value={${id}Data}`);
    
    // 根据出度生成 setValue 传递
    comp.outgoing.forEach(out => {
      const alias = comp.setValueAliases[out.target];
      const setter = `set${out.target}Data`;
      props.push(`${alias}={${setter}}`);
    });
    
    return `<${id} ${props.join(' ')} />`;
  }).join('\n      ');
  
  return `
import React, { useState } from 'react';
${Object.keys(componentMap).map(id => `import ${id} from './components/${id}';`).join('\n')}

const ${pageName} = () => {
  ${states}
  
  return (
    <div style={{ padding: '24px' }}>
      ${renders}
    </div>
  );
};

export default ${pageName};
`;
}
```

```javascript
// src/templates/service.js
function templateService(componentMap) {
  const apis = [];
  
  Object.keys(componentMap).forEach(id => {
    const comp = componentMap[id];
    const baseName = id.toLowerCase();
    
    if (comp.url.load) {

    }
    if (comp.url.query) {
      apis.push(`export const query${id} = (data) => request.post('${comp.url.query}', data);`);
    }
    if (comp.url.save) {
      apis.push(`export const save${id} = (data) => request.post('${comp.url.save}', data);`);
    }
  });
  
  return `
// 统一请求封装（根据项目实际情况调整）
import request from '@/utils/request';

${apis.join('\n')}
`;
}
```

```javascript
// bin/generate.js
#!/usr/bin/env node

const { program } = require('commander');
const chalk = require('chalk');
const path = require('path');
const { parseConfig } = require('../src/core/parser');
const { buildModel } = require('../src/core/modelBuilder');
const { render } = require('../src/core/renderer');
const { writeFiles } = require('../src/utils/fileWriter');

program
  .option('-c, --config <path>', '配置文件路径')
  .option('-o, --output <path>', '输出目录')
  .option('-n, --page-name <name>', '页面名称', 'Page')
  .parse(process.argv);

async function main() {
  const options = program.opts();
  
  console.log(chalk.blue('🚀 开始生成代码...'));
  
  try {
    // 1. 解析配置
    console.log(chalk.gray('📋 解析配置文件...'));
    const { componentMap, links } = parseConfig(options.config);
    console.log(chalk.green(`✅ 发现 ${Object.keys(componentMap).length} 个组件，${links.length} 条数据流`));

  } catch (error) {
```

```javascript
// src/utils/fileWriter.js
const fs = require('fs-extra');
const prettier = require('prettier');
const path = require('path');

function writeFiles(outputPath, output) {
  // 创建目录
  fs.ensureDirSync(outputPath);
  fs.ensureDirSync(path.join(outputPath, 'components'));
  
  // 写入 index.jsx
  fs.writeFileSync(
    path.join(outputPath, 'index.jsx'),
    formatCode(output.index)
  );
  
  // 写入各组件
  output.components.forEach(comp => {
    const compPath = path.join(outputPath, 'components', `${comp.name}.jsx`);
    fs.writeFileSync(compPath, formatCode(comp.code));
  });
  
  // 写入 service
  fs.writeFileSync(
    path.join(outputPath, 'service.js'),
    formatCode(output.service)
  );
}

function formatCode(code) {
  return prettier.format(code, {
    parser: 'babel',
    singleQuote: true,
    trailingComma: 'es5',
    printWidth: 100
  });
}
```

## 8. 关键实现细节

### 8.1 Form → Form 的特殊处理

当检测到 `formToForm` 流向时，需要从源 Form 的 `Fields` 中找出第一个 `type: "Select"` 的字段作为触发源：

```javascript
function findFirstSelectField(fields) {
  return fields.find(f => f.type === 'Select');
}

// 在渲染时特殊处理
if (behavior === 'formToForm') {
  const selectField = findFirstSelectField(comp.fields);
  if (!selectField) {
    throw new Error(`组件 ${comp.id} 要建立 Form→Form 连接，但未找到 Select 字段`);
  }
  // 记录该字段名，用于生成 onValuesChange 的监听逻辑
  comp.selectField = selectField.name;
}
```

### 8.2 Table → Table 的数据提取

当 Table 接收到 `record` 后，需要提取主键 ID 作为 `load` 请求参数。约定使用 `record.id` 或 `record.PROCESS_ID`：

```javascript
function getPrimaryKey(record) {
  return `${record?.id || record?.PROCESS_ID || ''}`;
}

// 生成的目标 Table 的 useEffect 代码
const loadEffect = `
useEffect(() => {
  if (props.value && Object.keys(props.value).length > 0) {
    const id = props.value.id || props.value.PROCESS_ID;
    if (id) {
      load${targetId}({ id }).then(res => {
        setInternalData(res.LIST || []);
      });
    }
  }
}, [props.value]);
`;
```
